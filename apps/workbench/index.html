# plaq_quantum/circuits.py - Bernstein-Vazirani Generator
def make_bernstein_vazirani_circuit(bitstring: str = "10110") -> QuantumCircuit:
    n_qubits = len(bitstring) + 1
    qc = QuantumCircuit(n_qubits, len(bitstring))
    # Initialize target ancilla qubit in |->
    qc.x(n_qubits - 1)
    qc.h(n_qubits - 1)
    # Hadamard transform input register
    for q in range(len(bitstring)):
        qc.h(q)
    return qc
